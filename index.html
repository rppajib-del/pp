Integrasi Google Spreadsheet

Simpan Otomatis Poin & Presensi Per-Tanggal dan Per-Kelas

Masukkan Web App URL dari Google Apps Script untuk otomatis menyinkronkan data.

Kode Apps Script Terbaru (Mendukung Poin + Presensi):

Salin kode script di bawah ini ke menu **Apps Script** Google Spreadsheet Anda:

function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  if (data.type === 'attendance') {
    // Sheet khusus presensi per kelas & tanggal (Contoh: Absen X Tekstil A - 2026-09-08)
    var sheetName = "Absen " + data.className + " (" + data.date + ")";
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) { sheet = ss.insertSheet(sheetName); }
    sheet.clear();
    sheet.appendRow(["SMKN 5 YOGYAKARTA - REKAP PRESENSI KELAS"]);
    sheet.appendRow(["Kelas:", data.className]);
    sheet.appendRow(["Tanggal:", data.date]);
    sheet.appendRow(["Materi:", data.material || "-"]);
    sheet.appendRow([]);
    sheet.appendRow(["No", "NISN", "Nama Murid", "Status Kehadiran", "Catatan"]);
    data.records.forEach(function(r, index) {
      sheet.appendRow([index + 1, r.nisn, r.name, r.status, r.note || "-"]);
    });
  } else {
    // Sheet Leaderboard Poin Per Kelas
    var sheet = ss.getSheetByName(data.className);
    if (!sheet) { sheet = ss.insertSheet(data.className); }
    sheet.clear();
    sheet.appendRow(["No", "NISN", "Nama Murid", "Total Poin", "Terakhir Diperbarui"]);
    data.students.forEach(function(s, index) {
      sheet.appendRow([index + 1, s.nisn, s.name, s.points, s.lastUpdated || new Date().toLocaleString("id-ID")]);
    });
  }
  return ContentService.createTextOutput(JSON.stringify({status: "success"}))
    .setMimeType(ContentService.MimeType.JSON);
}
